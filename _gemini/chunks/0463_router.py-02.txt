def reparse(p):
    try:
        return getattr(os.lstat(p), "st_reparse_tag", 0)
    except OSError:
        return -1


def is_link(p):
    return reparse(p) > 0


def parse_fm(path):
    """Return (name, description) from SKILL.md frontmatter. Pure stdlib."""
    try:
        with open(path, "r", encoding="utf-8", errors="replace") as f:
            txt = f.read()
    except OSError:
        return None, None
    if not txt.startswith("---"):
        return None, None
    end = txt.find("\n---", 3)
