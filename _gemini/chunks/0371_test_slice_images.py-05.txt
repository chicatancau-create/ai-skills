            with Image.open(io.BytesIO(adjusted)) as result:

        from gemini_watermark_remover import process_image

        with tempfile.TemporaryDirectory() as tmp:
            source = Path(tmp) / "image.jpg"
            image = Image.new("RGB", (256, 160), "red")
