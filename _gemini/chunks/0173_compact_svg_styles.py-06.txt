            for path, payload, _stats in prepared
        )

    print(json.dumps({
        "input": str(input_path),
        "inplace": bool(args.inplace),
        "file_count": len(prepared),
        "changed_files": changed_files,
        "styles": total.as_dict(),
    }, ensure_ascii=False))
    return 0
