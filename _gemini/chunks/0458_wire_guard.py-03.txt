            try:
                res = tick(True)
                key = tuple(res)
                if key != prev:
                    for r in res:
                        log(r)
                    prev = key
            except Exception as e:
                log("tick error: %s: %s" % (type(e).__name__, e))
            time.sleep(INTERVAL)
    apply_ = "--once" in args or "--apply" in args
    for r in tick(apply_):
        print(r)
    if not apply_:
        print("(report only; use --once to fix, --watch to keep it fixed)")

    main()
