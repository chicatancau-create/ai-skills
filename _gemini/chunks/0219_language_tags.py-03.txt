    parts = tag.split("-")
    index = 1
    if len(parts[0]) <= 3:
        while index < len(parts) and _EXTLANG_RE.fullmatch(parts[index]):
            index += 1
    if index < len(parts) and _SCRIPT_RE.fullmatch(parts[index]):
        return parts[index] in _RTL_SCRIPTS
    return parts[0] in _RTL_BASE_LANGUAGES
