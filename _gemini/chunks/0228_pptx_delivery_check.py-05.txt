        canonical = _PPT_SAFE_FONT_ALIASES.get(normalized, normalized)
        canonical = unicodedata.normalize(
            "NFKC",
            canonical,
        ).strip().casefold()
        if canonical.startswith("+") or canonical in _PPT_DELIVERY_SAFE_FONTS:
            continue
        unsafe.append(face)
    return unsafe
