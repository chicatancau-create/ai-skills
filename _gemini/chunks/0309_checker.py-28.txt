            if name in style_values:

            return element.get(name)

        def reference_kind(value: str | None) -> str:
            match = re.fullmatch(
                r'url\(\s*#([^)]+?)\s*\)',
                (value or '').strip(),
                re.IGNORECASE,
            )
            return definition_kinds.get(match.group(1), '') if match else ''
